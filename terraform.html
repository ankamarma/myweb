VM Deployment Guide on VMware ESXi Using Terraform

Environment & Tools

  • Hardware Platform: Cisco UCS C220 M4 Rack Server
  • Hypervisor: VMware ESXi
  • Infrastructure as Code: Terraform
  • Local Workstation OS: Windows with WSL2 (Ubuntu)
  • Source Image: Operating System OVA file (stored in the Ubuntu virtualbox-ova directory)

Step-by-Step Instructions

  1. Set up the environment: Open the Ubuntu WSL terminal on Windows.
  2. Install Terraform: Download and install Terraform inside your Ubuntu environment.
  3. Save your credentials (terraform.tfvars): Create this file to store your ESXi host IP, username, and password.
  4. Define your variables (variables.tf): Create this file to outline the setup options, like the operating system and VM count.
  5. Create the main configuration (main.tf): Create this file to define how to build the Ubuntu and Kali virtual machines on your host.
  6. Create the automation script (deploy.sh): Create this script to count existing virtual machines and add new ones safely without deleting the old ones.

terraform {
  required_providers {
    esxi = {
      source  = "josenk/esxi"
      version = "1.10.3"
    }
  }
}

provider "esxi" {
  esxi_hostname = var.esxi_hostname
  esxi_username = var.esxi_username
  esxi_password = var.esxi_password
}

resource "esxi_guest" "ubuntu_vms" {
  count              = lower(var.os_choice) == "ubuntu" ? var.vm_count : 0
  guest_name         = "ubuntu-node-${count.index + 1}"
  disk_store         = "automation-storage"
  ovf_source         = "/home/administrator/virtualbox-ova/noble-server-cloudimg-amd64.ova"
  memsize            = "2048"
  numvcpus           = "2"
  power              = "on"
  resource_pool_name = "/"

  network_interfaces {
    virtual_network = "VM Network"
  }

  guestinfo = {
    "userdata" = base64encode(<<-EOF
      #cloud-config
      user: admin
      password: automation@12
      chpasswd: { expire: False }
      ssh_pwauth: True
    EOF
    )
    "userdata.encoding" = "base64"
  }
}
